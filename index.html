        // Smart Advice logic update for teddy bear mascot expression

        function updateMascotAdvice(todayDebit, dailyLimit, todaySafeSpend, remainingMonthlyBudget, progressPct) {

            const adviceBox = document.getElementById('smart-advice-text');

            const mascotEmoji = document.getElementById('mascot-mood-emoji');

 

            if (todayDebit > dailyLimit) {

                if (mascotEmoji) mascotEmoji.innerText = "🧸💔";

                adviceBox.innerHTML = `<span class="text-accent-rose font-bold">Oh dear! Overspent by RM ${(todayDebit - dailyLimit).toFixed(2)} today.</span> Teddy adjusted tomorrow's limit for you~`;

            } else if (todayDebit > (dailyLimit * 0.7)) {

                if (mascotEmoji) mascotEmoji.innerText = "🧸🍯";

                adviceBox.innerHTML = `You've used <span class="text-amber-700 font-bold">${progressPct.toFixed(0)}%</span> of today's limit. Mind the extra honey snacks & treats! ☕`;

            } else {

                if (mascotEmoji) mascotEmoji.innerText = "🧸✨";

                adviceBox.innerHTML = `Great job staying budget-smart! You have <span class="text-accent-emerald font-bold">RM ${todaySafeSpend.toFixed(2)}</span> left today. Monthly stash: RM ${remainingMonthlyBudget.toFixed(2)} 💖`;

            }

        }

        function renderDashboard() {

            // ... existing calculation code ...

            const todayStr = new Date().toISOString().split('T')[0];

            const now = new Date();

            const currentMonth = now.getMonth();

            const currentYear = now.getFullYear();

 

            const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();

            const daysRemaining = daysInMonth - now.getDate() + 1;

 

            const monthlyTrx = appState.transactions.filter(t => {

                const d = new Date(t.timestamp);

                return d.getMonth() === currentMonth && d.getFullYear() === currentYear;

            });

 

            const monthlyDebit = monthlyTrx.filter(t => t.type === 'debit').reduce((sum, t) => sum + t.amount, 0);

            const remainingMonthlyBudget = Math.max(0, appState.monthlyBudget - monthlyDebit);

            const dailyLimit = daysRemaining > 0 ? (remainingMonthlyBudget / daysRemaining) : 0;

 

            const todayTrx = appState.transactions.filter(t => t.date === todayStr);

            const todayDebit = todayTrx.filter(t => t.type === 'debit').reduce((sum, t) => sum + t.amount, 0);

            const todayCredit = todayTrx.filter(t => t.type === 'credit').reduce((sum, t) => sum + t.amount, 0);

 

            const todaySafeSpend = Math.max(0, dailyLimit - todayDebit);

 

            document.getElementById('daily-remaining').innerText = todaySafeSpend.toFixed(2);

            document.getElementById('today-spent-text').innerText = `RM ${todayDebit.toFixed(2)}`;

            document.getElementById('daily-limit-text').innerText = `Limit: RM ${dailyLimit.toFixed(2)}`;

 

            const progressPct = dailyLimit > 0 ? Math.min(100, (todayDebit / dailyLimit) * 100) : 100;

            const progressBar = document.getElementById('daily-progress-bar');

            progressBar.style.width = `${progressPct}%`;

           

            if (progressPct > 90) {

                progressBar.className = "bg-accent-rose h-2 rounded-full transition-all duration-500";

            } else if (progressPct > 60) {

                progressBar.className = "bg-accent-amber h-2 rounded-full transition-all duration-500";

            } else {

                progressBar.className = "bg-accent-emerald h-2 rounded-full transition-all duration-500";

            }

 

            updateMascotAdvice(todayDebit, dailyLimit, todaySafeSpend, remainingMonthlyBudget, progressPct);

 

            document.getElementById('today-debit').innerText = `RM ${todayDebit.toFixed(2)}`;

            document.getElementById('debit-count').innerText = `${todayTrx.filter(t => t.type === 'debit').length} spent 🛍️`;

            document.getElementById('today-credit').innerText = `RM ${todayCredit.toFixed(2)}`;

            document.getElementById('credit-count').innerText = `${todayTrx.filter(t => t.type === 'credit').length} received 🎉`;

 

            renderPeriodSummary();

            renderTransactionList();

        }

 

        function closeManualModal() {

            document.getElementById('manual-modal').classList.add('hidden');

        }

 

        function saveManualTransaction() {

            const amount = parseFloat(document.getElementById('manual-amount').value);

            const title = document.getElementById('manual-title-input').value.trim() || (appState.pendingType === 'debit' ? 'Daily Expense' : 'Wallet Reload');

            const source = document.getElementById('manual-source').value;

            const todayStr = new Date().toISOString().split('T')[0];

 

            if (isNaN(amount) || amount <= 0) return;

 

            appState.transactions.push({

                id: Date.now(),

                source: source,

                type: appState.pendingType,

                title: title,

                amount: amount,

                date: todayStr,

                timestamp: new Date().toISOString()

            });

 

            saveToLocalStorage();

            renderDashboard();

            closeManualModal();

        }

 

        function clearAllTransactions() {

            appState.transactions = [];

            saveToLocalStorage();

            renderDashboard();

        }

 

        function saveToLocalStorage() {

            localStorage.setItem('auto_expense_app_data', JSON.stringify(appState));

        }

 

        function loadFromLocalStorage() {

            const data = localStorage.getItem ? localStorage.getItem('auto_expense_app_data') : null;

            if (data) {

                try {

                    appState = JSON.parse(data);

                } catch(e) {

                    console.error("Failed to parse stored data", e);

                }

            }

        }
